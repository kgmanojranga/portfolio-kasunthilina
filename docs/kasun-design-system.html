Reference · 01

Animations

All motion uses these three keyframes. Copy directly into your stylesheet. 

/* Entry — used on all section reveals */
@keyframes fadeUp {
  from { opacity: 0; transform: translateY(20px); }
  to   { opacity: 1; transform: translateY(0); }
}

/* Scroll indicator line drop */
@keyframes scrollDrop {
  0%   { transform: scaleY(0); transform-origin: top; opacity: 0; }
  50%  { transform: scaleY(1); transform-origin: top; opacity: 1; }
  100% { transform: scaleY(1); transform-origin: bottom; opacity: 0; }
}

/* Status dot pulse */
@keyframes pulse-dot {
  0%,100% { box-shadow: 0 0 0 3px rgba(74,222,128,0.18); }
  50%     { box-shadow: 0 0 0 6px rgba(74,222,128,0.06); }
}

/* Stagger pattern for hero */
.hero-eyebrow   { animation: fadeUp 0.7s 0.20s both; }
.hero-name      { animation: fadeUp 0.7s 0.35s both; }
.hero-tagline   { animation: fadeUp 0.7s 0.50s both; }
.hero-btns      { animation: fadeUp 0.7s 0.65s both; }
.hero-stats     { animation: fadeUp 0.7s 0.80s both; }
.g-card:nth-child(1) { animation: fadeUp 0.7s 0.40s both; }
.g-card:nth-child(2) { animation: fadeUp 0.7s 0.55s both; }
.g-card:nth-child(3) { animation: fadeUp 0.7s 0.70s both; }

/* Scroll-triggered reveal (via IntersectionObserver) */
.reveal { opacity: 0; transform: translateY(24px); transition: opacity 0.6s ease, transform 0.6s ease; }
.reveal.visible { opacity: 1; transform: none; }

/* Respect reduced motion */
@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after { animation-duration: 0.01ms !important; transition-duration: 0.01ms !important; }
}
Reference · 02

CSS Variables — Full Reference

Copy this block into your globals.css as the first thing after your font imports. 

:root {
  /* ── Backgrounds ── */
  --bg-page:      #0a0c10;
  --bg-surface:   #0f1117;
  --bg-raised:    #161820;
  --bg-card:      rgba(255,255,255,0.055);

  /* ── Glass ── */
  --glass-bg:          rgba(255,255,255,0.055);
  --glass-bg-hover:    rgba(255,255,255,0.085);
  --glass-border:      rgba(255,255,255,0.10);
  --glass-border-hover:rgba(255,255,255,0.18);
  --glass-blur:        blur(20px) saturate(160%);

  /* ── Text ── */
  --text-primary:   #ffffff;
  --text-secondary: rgba(255,255,255,0.60);
  --text-muted:     rgba(255,255,255,0.32);
  --text-hint:      rgba(255,255,255,0.18);

  /* ── Accent ── */
  --accent-primary:   #1C3557;
  --accent-secondary: #3B82C4;
  --accent-tint:      rgba(59,130,196,0.15);

  /* ── Semantic ── */
  --success:      #4ade80;
  --success-tint: rgba(74,222,128,0.10);
  --warning:      #f59e0b;
  --error:        #f87171;

  /* ── Borders ── */
  --border-subtle:   rgba(255,255,255,0.07);
  --border-default:  rgba(255,255,255,0.10);
  --border-emphasis: rgba(255,255,255,0.20);

  /* ── Shadows ── */
  --shadow-sm: 0 1px 3px rgba(0,0,0,0.4);
  --shadow-md: 0 4px 16px rgba(0,0,0,0.5);
  --shadow-lg: 0 12px 40px rgba(0,0,0,0.6);

  /* ── Radius ── */
  --r-xs:   4px;  --r-sm: 8px;
  --r-md:   12px; --r-lg: 16px;
  --r-xl:   20px; --r-pill: 999px;

  /* ── Spacing ── */
  --sp-1:4px;  --sp-2:8px;  --sp-3:12px;
  --sp-4:16px; --sp-5:20px; --sp-6:24px;
  --sp-8:32px; --sp-10:40px; --sp-12:48px;

  /* ── Transitions ── */
  --t-fast: 150ms ease;
  --t-base: 250ms ease;
  --t-slow: 400ms ease;

  /* ── Font families ── */
  --font-display: 'Plus Jakarta Sans', sans-serif;
  --font-body:    'Inter', sans-serif;
  --font-mono:    'JetBrains Mono', monospace;
}
